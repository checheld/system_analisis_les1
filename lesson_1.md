```mermaid

    "fontFamily": "Inter, Arial, sans-serif"
  },
  "sequence": {
    "showSequenceNumbers": true,
    "diagramMarginX": 14
  }
}}%%
sequenceDiagram
    actor K as Клиент
    participant U as API упр-я заказами
    participant BD@{"type": "database"} as БД
    participant S as API службы доставки
    K->>U: Создание заказа
    U->>U: Проверка валидации входных данных
    alt Корректная валидация  входных данных
      U->>BD: Сохранение заказа в БД
      activate BD
      BD->>U: Заказ создан со статусом "создан"
      deactivate BD
      loop 5 попыток создания слота в СД c интервалами 500 ms, 1s, 2s, 4s.
        U->>S: Создание слота в СД
        activate S
        S->>U: Ответ СД (200 OK / timeout / network error)
        deactivate S
      end
      alt Cлот СД зарезервирован
      U->>BD: Обновление заказа в БД
      activate BD
      BD->>U: Заказ обновлен, статус изменен на "подтвержден"
      deactivate BD
      U->>K: 200 OK Статус заказа "Подтвержден"
        else слот СД не зарезервирован
        U->>K: 200 OK Статус заказа "Создан"
      end
    else Некорректная валидация  входных данных
    U->>K: 400 Bad request
    end
```

@startuml
actor "Клиент" as K
participant "API упр-я заказами" as U

    U ->> BD : Сохранение заказа в БД
    activate BD

**3 Таблица маппинга**

| Поле нашей системы   | Поле внешнего API                              | Логика преобразования                                                 |
| -------------------- | ---------------------------------------------- | --------------------------------------------------------------------- |
| id                   | externalOrderId                                | Меняется название поля                                                |
| deliveryId           | id                                             | Меняется название поля                                                |
| city                 | destination.city                               | Город добавляется в объект "destination"                              |
| address              | destination.address                            | Адресс добавляется в объект "destination"                             |
| firstname + lastname | fullname                                       | Объединение двух полей в одно                                         |
| date                 | deliveryWindow.date                            | Дата добавляется в объект "deliveryWindow"                            |
| time                 | deliveryWindow.timeFrom, deliveryWindow.timeTo | Время добавляется в объект "deliveryWindow" и разбивается на два поля |
